```python
from funasr import AutoModel
from .asr_engine import ASREngine, ASRResult, ASRSegment

class FunASREngine(ASREngine):
    """FunASR本地语音识别引擎"""

    def __init__(self, model_name: str = "paraformer-zh", device: str = "cuda"):
        super().__init__()
        self.model = AutoModel(
            model=model_name,
            vad_model="fsmn-vad",
            punc_model="ct-punc",
            device=device
        )

    def transcribe(self, audio: np.ndarray, sample_rate: int = 16000) -> ASRResult:
        result = self.model.generate(input=audio, batch_size_s=300)
        return self._convert_result(result)

    def transcribe_file(self, file_path: str) -> ASRResult:
        result = self.model.generate(input=file_path, batch_size_s=300)
        return self._convert_result(result)
```

```python
# backend/src/perception/funasr_streaming.py
from funasr import AutoModel

class FunASRStreaming:
    """FunASR流式语音识别"""

    def __init__(self, chunk_size: list = [0, 10, 5]):
        self.model = AutoModel(model="paraformer-zh-streaming")
        self.chunk_size = chunk_size  # [0,10,5] = 600ms延迟
        self.cache = {}

    async def process_chunk(self, audio_chunk: bytes) -> Optional[str]:
        """处理音频块，返回识别结果"""
        result = self.model.generate(
            input=audio_chunk,
            cache=self.cache,
            is_final=False,
            chunk_size=self.chunk_size
        )
        return result[0]["text"] if result else None
```

```python
# backend/src/evaluation/ollama_client.py
import requests
from typing import Optional, Generator

class OllamaClient:
    """Ollama本地LLM客户端"""

    def __init__(
        self,
        base_url: str = "http://localhost:11434",
        model: str = "qwen2.5:14b",
        timeout: int = 60
    ):
        self.base_url = base_url
        self.model = model
        self.timeout = timeout

    def generate(self, prompt: str, stream: bool = False) -> str:
        """生成文本响应"""
        response = requests.post(
            f"{self.base_url}/api/generate",
            json={"model": self.model, "prompt": prompt, "stream": stream},
            timeout=self.timeout
        )
        return response.json()["response"]

    def chat(self, messages: list, stream: bool = False) -> str:
        """对话模式"""
        response = requests.post(
            f"{self.base_url}/api/chat",
            json={"model": self.model, "messages": messages, "stream": stream},
            timeout=self.timeout
        )
        return response.json()["message"]["content"]

    def is_available(self) -> bool:
        """检查Ollama服务是否可用"""
        try:
            requests.get(f"{self.base_url}/api/tags", timeout=5)
            return True
        except:
            return False
```
